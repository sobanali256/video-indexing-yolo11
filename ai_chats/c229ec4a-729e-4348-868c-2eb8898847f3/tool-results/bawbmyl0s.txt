+"""Stage-0 decode profile (CPU only, no model): how fast can one process decode, and what does BGR
+conversion cost? Decides whether decoding needs threads or worker processes once sampling cuts model work.
+
+    python tools/profile_decode.py --input /content/subset/tune --videos 5
+
+Modes per FFmpeg thread count (0 = FFmpeg's automatic choice):
+  decode   decode + timestamp only (lower bound: work that sampling cannot skip)
+  bgr_all  decode + BGR conversion of every frame (what brute force does)
+  sample5  decode + BGR conversion of 5 frames per second (vidx.decode with sample=5)
+"""
+import argparse
+import os
+import sys
+import time
+from pathlib import Path
+
+sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
+
+from vidx.decode import Timestamper, _open, pick_video_stream  # noqa: E402
+from vidx.sampling import sample_groups  # noqa: E402
+from vidx.source import iter_rows, list_inputs  # noqa: E402
+
+
+def run_mode(data, mode, threads):
+    """-> (decoded frames, converted frames, seconds)."""
+    t0 = time.perf_counter()
+    container = _open(data)
+    decoded = converted = 0
+    try:
+        stream = pick_video_stream(container)
+        if stream is None:
+            return 0, 0, 0.0
+        stream.thread_type = "AUTO"
+        if threads:
+            stream.thread_count = threads
+        stamper = Timestamper(stream.time_base, stream.guessed_rate or stream.average_rate)
+
+        def stamped():
+            nonlocal decoded
+            for frame in container.decode(stream):
+                decoded += 1
+                yield frame, stamper.stamp(frame.pts)
+
+        frames = stamped()
+        if mode == "sample5":
+            frames = sample_groups(frames, lambda x: x[1], 5)
+        for frame, _ in frames:
+            if mode != "decode":
+                frame.to_ndarray(format="bgr24")
+                converted += 1
+    finally:
+        container.close()
+    return decoded, converted, time.perf_counter() - t0
+
+
+def main(argv=None):
+    ap = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
+    ap.add_argument("--input", required=True)
+    ap.add_argument("--videos", type=int, default=5)
+    ap.add_argument("--threads", default="0,1,2,4", help="FFmpeg thread counts to try (0 = auto)")
+    ap.add_argument("--modes", default="decode,bgr_all,sample5")
+    args = ap.parse_args(argv)
+

+"""Per-second frame sampling. Standard library only.
+
+Used by both the decoder (vidx.decode) and the offline simulator (tools/simulate_sampling.py),
+so the simulator's scores predict the pipeline exactly: same grouping, same selection.
+
+A "group" is a run of consecutive frames whose int(t) is the same second. A group closes as soon
+as a frame of a different second arrives (also when timestamps go backwards), so a second can, in
+rare broken files, produce two groups; both the simulator and the pipeline then sample each group.
+Every group yields at least one frame, so every second that holds a frame keeps an output line.
+"""
+
+
+def pick(n, k):
+    """Indices of the frames to keep from a group of n frames: k evenly spaced ones.
+
+    k <= 0 (sampling off) or n <= k -> all n frames. Otherwise the centres of k equal slices:
+    floor((j + 0.5) * n / k) for j in 0..k-1 (strictly increasing because n > k).
+    """
+    if k <= 0 or n <= k:
+        return list(range(n))
+    return [(2 * j + 1) * n // (2 * k) for j in range(k)]
+
+
+def second_of(t):
+    """Second index of a timestamp (Fraction or int, >= 0); same rule as vidx.aggregate."""
+    return int(t // 1)
+
+
+def iter_groups(items, t_of):
+    """Yield lists of consecutive items that share a second (see module docstring).
+
+    If the iterable raises (decode failure), the group collected so far is yielded first and then
+    the exception is re-raised, so frames decoded before the error are never lost.
+    """
+    group = []
+    current = None
+    it = iter(items)
+    while True:
+        try:
+            item = next(it)
+        except StopIteration:
+            break
+        except Exception:
+            if group:
+                yield group
+            raise
+        s = second_of(t_of(item))
+        if group and s != current:
+            yield group
+            group = []
+        current = s
+        group.append(item)
+    if group:
+        yield group
+
+
+def sample_groups(items, t_of, k):
+    """Yield the selected items (pick(n, k) of each group), group by group, in input order."""
+    for group in iter_groups(items, t_of):
+        yield from (group[i] for i in pick(len(group), k))
