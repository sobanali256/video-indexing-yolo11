     1	"""Batch vs single-frame inference, and classes-filter vs no-filter (needs the model, i.e. a GPU box).
     2	
     3	    python tools/batch_agreement.py --input finevideo --frames 1000
     4	
     5	Targets: batch vs single >= 0.999 identical frames; filter vs no filter identical.
     6	"""
     7	import argparse
     8	import sys
     9	from pathlib import Path
    10	
    11	sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
    12	
    13	from vidx.decode import decode_video  # noqa: E402
    14	from vidx.model import infer, load_model  # noqa: E402
    15	from vidx.source import iter_rows  # noqa: E402
    16	
    17	
    18	def collect(input_dir, n):
    19	    """Up to n frames: every 15th frame of the first videos, so several videos are used."""
    20	    images = []
    21	    for path in sorted(Path(input_dir).glob("*.parquet")):
    22	        for _, data, err in iter_rows(path):
    23	            if err:
    24	                continue
    25	            try:
    26	                for i, f in enumerate(decode_video(data)):
    27	                    if i % 15 == 0:
    28	                        images.append(f.image)
    29	                        if len(images) >= n:
    30	                            return images
    31	            except Exception as e:
    32	                print("decode problem, skipping rest of video:", e, file=sys.stderr)
    33	    return images
    34	
    35	
    36	def main():
    37	    ap = argparse.ArgumentParser()
    38	    ap.add_argument("--input", required=True)
    39	    ap.add_argument("--frames", type=int, default=1000)
    40	    ap.add_argument("--batch", type=int, default=32)
    41	    args = ap.parse_args()
    42	    model = load_model()
    43	    images = collect(args.input, args.frames)
    44	    print(f"{len(images)} frames collected")
    45	
    46	    single = [infer(model, [im])[0] for im in images]
    47	    batched = []
    48	    i = 0
    49	    while i < len(images):  # same flush rule as run.py: a batch holds one frame shape only
    50	        j = i + 1
    51	        while j < len(images) and j - i < args.batch and images[j].shape == images[i].shape:
    52	            j += 1
    53	        batched += infer(model, images[i:j])
    54	        i = j
    55	    unfiltered = [infer(model, [im], classes=None)[0] for im in images]
    56	
    57	    def same(a, b):
    58	        return sum(1 for x, y in zip(a, b) if x[0] == y[0]) / max(len(a), 1)
    59	
    60	    def max_conf_gap(a, b):
    61	        return max((abs(p - q) for x, y in zip(a, b) for p, q in zip(x[1], y[1])), default=0.0)
    62	
    63	    print(f"batch vs single:        identical counts {same(single, batched):.4f} (need >= 0.999), "
    64	          f"max conf gap {max_conf_gap(single, batched):.5f}")
    65	    print(f"classes filter vs none: identical counts {same(single, unfiltered):.4f} (need 1.0), "
    66	          f"max conf gap {max_conf_gap(single, unfiltered):.5f}")
    67	
    68	
    69	if __name__ == "__main__":
    70	    main()

     1	"""Check a results file against the output format. Exit 1 if anything is wrong.
     2	
     3	    python tools/check_schema.py results.jsonl
     4	"""
     5	import json
     6	import sys
     7	from collections import defaultdict
     8	from pathlib import Path
     9	
    10	sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
    11	
    12	from vidx.aggregate import OBJECTS  # noqa: E402
    13	
    14	
    15	def check(path):
    16	    problems = []
    17	    seconds = defaultdict(list)
    18	    errors = defaultdict(int)
    19	    with open(path, encoding="utf-8") as f:
    20	        for n, raw in enumerate(f, 1):
    21	            if not raw.strip():
    22	                continue
    23	            try:
    24	                d = json.loads(raw)
    25	            except ValueError:
    26	                problems.append(f"line {n}: not valid JSON")
    27	                continue
    28	            if not isinstance(d, dict):
    29	                problems.append(f"line {n}: not a JSON object")
    30	                continue
    31	            video = d.get("video")
    32	            if not isinstance(video, str) or ":" not in video:
    33	                problems.append(f"line {n}: bad video id {video!r}")
    34	                continue
    35	            if "error" in d:
    36	                if set(d) != {"video", "error"} or not isinstance(d["error"], str):
    37	                    problems.append(f"line {n}: malformed error line")
    38	                errors[video] += 1
    39	                continue
    40	            if set(d) != {"video", "second", "objects", "people"}:
    41	                problems.append(f"line {n}: wrong keys {sorted(d)}")
    42	                continue
    43	            sec, objs, ppl = d["second"], d["objects"], d["people"]
    44	            if not isinstance(sec, int) or isinstance(sec, bool) or sec < 0:
    45	                problems.append(f"line {n}: bad second {sec!r}")
    46	            if not isinstance(ppl, int) or isinstance(ppl, bool) or ppl < 0:
    47	                problems.append(f"line {n}: bad people {ppl!r}")
    48	            if not isinstance(objs, list) or any(o not in OBJECTS for o in objs):
    49	                problems.append(f"line {n}: bad objects {objs!r}")
    50	            elif objs != [o for o in OBJECTS if o in objs]:
    51	                problems.append(f"line {n}: objects not unique/in canonical order {objs!r}")
    52	            seconds[video].append(sec)
    53	    for video, secs in seconds.items():
    54	        if sorted(secs) != list(range(len(secs))):
    55	            problems.append(f"{video}: seconds not contiguous from 0 (got {len(secs)} lines, max {max(secs)})")
    56	    for video, n in errors.items():
    57	        if n > 1:
    58	            problems.append(f"{video}: {n} error lines")
    59	    print(f"{len(seconds)} videos with seconds, {len(errors)} with error lines")
    60	    return problems
    61	
    62	
    63	def main():
    64	    if len(sys.argv) != 2:
    65	        raise SystemExit(__doc__)
    66	    problems = check(sys.argv[1])
    67	    for p in problems[:50]:
    68	        print("PROBLEM:", p)
    69	    if len(problems) > 50:
    70	        print(f"... and {len(problems) - 50} more")
    71	    print("schema OK" if not problems else f"{len(problems)} problems")
    72	    return 1 if problems else 0
    73	
    74	
    75	if __name__ == "__main__":
    76	    sys.exit(main())

     1	"""Score PyAV decoding against cv2.VideoCapture on N videos (needs the model, i.e. a GPU box).
     2	
     3	    python tools/decoder_agreement.py --input finevideo --n 20
     4	
     5	Both decoders go through the same model and aggregation; the PyAV result is the
     6	reference. Agreement below ~0.99 means: ask the user which decoder is the reference.
     7	"""
     8	import argparse
     9	import os
    10	import sys
    11	import tempfile
    12	from fractions import Fraction
    13	from pathlib import Path
    14	
    15	sys.path.insert(0, str(Path(__file__).resolve().parent.parent))
    16	
    17	import score  # noqa: E402
    18	from run import Timers, process_frames, video_lines  # noqa: E402
    19	from vidx.decode import Frame, decode_video  # noqa: E402
    20	from vidx.model import infer, load_model  # noqa: E402
    21	from vidx.source import iter_rows  # noqa: E402
    22	
    23	
    24	def cv2_frames(data):
    25	    import cv2
    26	
    27	    fd, tmp = tempfile.mkstemp(suffix=".mp4")
    28	    try:
    29	        with os.fdopen(fd, "wb") as f:
    30	            f.write(memoryview(data))
    31	        cap = cv2.VideoCapture(tmp)  # applies rotation metadata on its own
    32	        first = None
    33	        while True:
    34	            ok, image = cap.read()
    35	            if not ok:
    36	                break
    37	            ms = cap.get(cv2.CAP_PROP_POS_MSEC)
    38	            t = Fraction(ms / 1000.0).limit_denominator(1_000_000)
    39	            first = t if first is None else first
    40	            yield Frame(image, max(t - first, Fraction(0)), None, 0)
    41	        cap.release()
    42	    finally:
    43	        os.remove(tmp)
    44	
    45	
    46	def lines_for(video_id, frames, infer_fn, batch):
    47	    records, error = process_frames(frames, infer_fn, batch, Timers())
    48	    return video_lines(video_id, records, error), len(records)
    49	
    50	
    51	def main():
    52	    ap = argparse.ArgumentParser()
    53	    ap.add_argument("--input", required=True)
    54	    ap.add_argument("--n", type=int, default=20)
    55	    ap.add_argument("--batch", type=int, default=32)
    56	    args = ap.parse_args()
    57	    model = load_model()
    58	
    59	    def infer_fn(images):
    60	        return infer(model, images)
    61	
    62	    ref_lines, cand_lines = [], []
    63	    done = 0
    64	    for path in sorted(Path(args.input).glob("*.parquet")):
    65	        for video_id, data, err in iter_rows(path):
    66	            if err:
    67	                continue
    68	            a, na = lines_for(video_id, decode_video(data), infer_fn, args.batch)
    69	            b, nb = lines_for(video_id, cv2_frames(data), infer_fn, args.batch)
    70	            print(f"{video_id}: pyav {na} frames, cv2 {nb} frames")
    71	            ref_lines += a
    72	            cand_lines += b
    73	            done += 1
    74	            if done >= args.n:
    75	                break
    76	        if done >= args.n:
    77	            break
    78	    m = score.compute(score.parse_lines(ref_lines), score.parse_lines(cand_lines))
    79	    score.report(m)
    80	    print(f"decoder agreement (micro F1): {m['micro_f1']:.4f}  (< 0.99: ask the user which decoder is the reference)")

    84	    main()

# video-indexing-yolo11

Per-second object indexing of FineVideo with Ultralytics YOLO11l (person, car, dog, laptop, cell phone + people count), optimised for speed.

> Work in progress: the current `run.py` is the correct every-frame baseline. The performance section will be filled in once the speed-ups are validated.

## Install (Ubuntu 22.04, Python 3.11, CUDA 12)
```
sudo apt-get install -y libgl1 libglib2.0-0     # runtime libs for OpenCV
python3.11 -m venv .venv && source .venv/bin/activate
pip install -r requirements.txt
```

## Prepare (untimed, needs internet)
Downloads `yolo11l.pt` next to `run.py` and loads it once to check it works.
```
python prepare.py
```

## Run (no internet needed)
```
python run.py --input <folder of .parquet files> --output results.jsonl
```
Optional flags: `--batch 32`, `--resume` (continue an interrupted run via `results.jsonl.done`), `--limit-videos N`, `--dump-frames DIR` (per-frame npz for offline experiments). Stage timings (decode / infer / aggregate / total) are printed to stderr.

## Score a result against a reference
```
python score.py reference.jsonl candidate.jsonl
```

## Tests (CPU, standard library only)
```
python -m unittest discover -s tests -v
```

Colab L4 workflow: see `colab/SETUP.md`.
