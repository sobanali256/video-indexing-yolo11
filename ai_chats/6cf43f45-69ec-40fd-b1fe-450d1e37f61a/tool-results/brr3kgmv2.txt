import contextlib
import io
import os
import sys
import unittest

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

import score  # noqa: E402


def sec(video, s, objects=(), people=0):
    return {"video": video, "second": s, "objects": list(objects), "people": people}


def metrics(ref, cand):
    return score.compute(score.parse_lines(ref), score.parse_lines(cand))


class ScoreTest(unittest.TestCase):
    def test_identical(self):
        ref = [sec("v:0", 0, ["person"], 2), sec("v:0", 1, ["car", "dog"], 0), sec("v:1", 0, [], 0)]
        m = metrics(ref, list(ref))
        self.assertEqual(m["micro_f1"], 1.0)
        self.assertEqual(m["macro_f1"], 1.0)
        self.assertEqual(m["people_exact"], 1.0)
        self.assertEqual(m["people_within1"], 1.0)
        with contextlib.redirect_stdout(io.StringIO()):
            self.assertTrue(score.report(m))

    def test_both_empty_is_perfect(self):
        ref = [sec("v:0", 0), sec("v:0", 1)]
        m = metrics(ref, list(ref))
        self.assertEqual(m["micro_f1"], 1.0)
        self.assertEqual(m["video_f1"]["v:0"], 1.0)

    def test_missing_seconds_count_as_empty(self):
        ref = [sec("v:0", 0, ["car"], 1), sec("v:0", 1, [], 0)]
        cand = [sec("v:0", 1, [], 0)]
        m = metrics(ref, cand)
        self.assertEqual(m["missing_seconds"], 1)
        self.assertEqual(m["micro_f1"], 0.0)  # tp 0, fn 1
        self.assertEqual(m["people_exact"], 0.5)  # 1 vs predicted 0 is wrong; empty second matches
        self.assertEqual(m["people_within1"], 1.0)

    def test_extra_seconds_are_false_positives(self):
        ref = [sec("v:0", 0, ["car"], 0)]
        cand = [sec("v:0", 0, ["car"], 0), sec("v:0", 1, ["dog", "car"], 3)]
        m = metrics(ref, cand)
        self.assertEqual(m["extra_seconds"], 1)
        self.assertAlmostEqual(m["micro_f1"], 2 * 1 / (2 * 1 + 2 + 0))
        self.assertEqual(m["people_exact"], 1.0)  # people only judged on reference seconds

    def test_people_tolerance(self):
        ref = [sec("v:0", i, [], 2) for i in range(4)]
        cand = [sec("v:0", 0, [], 2), sec("v:0", 1, [], 3), sec("v:0", 2, [], 1), sec("v:0", 3, [], 5)]
        m = metrics(ref, cand)
        self.assertEqual(m["people_exact"], 0.25)
        self.assertEqual(m["people_within1"], 0.75)

    def test_thresholds_fail(self):
        ref = [sec("v:0", i, ["car"], 0) for i in range(10)]
        cand = [sec("v:0", i, [] if i < 2 else ["car"], 0) for i in range(10)]
        with contextlib.redirect_stdout(io.StringIO()):
            self.assertFalse(score.report(metrics(ref, cand)))

    def test_error_lines_counted_separately(self):
        ref = [sec("v:0", 0, ["car"], 0), {"video": "v:1", "error": "bad"}]
        cand = [sec("v:0", 0, ["car"], 0), {"video": "v:2", "error": "bad"}]
        m = metrics(ref, cand)
        self.assertEqual((m["ref_errors"], m["cand_errors"], m["error_mismatch"]), (1, 1, 2))
        self.assertEqual(m["micro_f1"], 1.0)

    def test_empty_files(self):
        m = metrics([], [])
        self.assertEqual((m["micro_f1"], m["people_exact"], m["people_within1"]), (1.0, 1.0, 1.0))


if __name__ == "__main__":
    unittest.main()
"""Check a results file against the output format. Exit 1 if anything is wrong.

    python tools/check_schema.py results.jsonl
"""
import json
import sys
from collections import defaultdict
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from vidx.aggregate import OBJECTS  # noqa: E402


def check(path):
    problems = []
    seconds = defaultdict(list)
    errors = defaultdict(int)
    with open(path, encoding="utf-8") as f:
        for n, raw in enumerate(f, 1):
            if not raw.strip():
                continue
            try:
                d = json.loads(raw)
            except ValueError:
                problems.append(f"line {n}: not valid JSON")
                continue
            if not isinstance(d, dict):
                problems.append(f"line {n}: not a JSON object")
                continue
            video = d.get("video")
            if not isinstance(video, str) or ":" not in video:
                problems.append(f"line {n}: bad video id {video!r}")
                continue
            if "error" in d:
                if set(d) != {"video", "error"} or not isinstance(d["error"], str):
                    problems.append(f"line {n}: malformed error line")
                errors[video] += 1
                continue
            if set(d) != {"video", "second", "objects", "people"}:
                problems.append(f"line {n}: wrong keys {sorted(d)}")
                continue
            sec, objs, ppl = d["second"], d["objects"], d["people"]
            if not isinstance(sec, int) or isinstance(sec, bool) or sec < 0:
                problems.append(f"line {n}: bad second {sec!r}")
            if not isinstance(ppl, int) or isinstance(ppl, bool) or ppl < 0:
                problems.append(f"line {n}: bad people {ppl!r}")
            if not isinstance(objs, list) or any(o not in OBJECTS for o in objs):
                problems.append(f"line {n}: bad objects {objs!r}")
            elif objs != [o for o in OBJECTS if o in objs]:
                problems.append(f"line {n}: objects not unique/in canonical order {objs!r}")
            seconds[video].append(sec)
    for video, secs in seconds.items():
        if sorted(secs) != list(range(len(secs))):
            problems.append(f"{video}: seconds not contiguous from 0 (got {len(secs)} lines, max {max(secs)})")
    for video, n in errors.items():
        if n > 1:
            problems.append(f"{video}: {n} error lines")
    print(f"{len(seconds)} videos with seconds, {len(errors)} with error lines")
    return problems


def main():
    if len(sys.argv) != 2:
        raise SystemExit(__doc__)
    problems = check(sys.argv[1])
    for p in problems[:50]:
        print("PROBLEM:", p)
    if len(problems) > 50:
        print(f"... and {len(problems) - 50} more")
    print("schema OK" if not problems else f"{len(problems)} problems")
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main())
# Colab L4 setup (VS Code Colab extension)

The ready-made notebook `colab/baseline.ipynb` contains every step below as cells.

How the extension works: you open a notebook in VS Code and pick a Colab server as its kernel; cells run on
Google's VM, not on your machine. Local files are **not** synced, so the code reaches Colab via `git clone`.
Shell commands go in cells as `!cmd` or `%%bash`. Every `!` line is a fresh shell, so `source activate` does
not persist; always call the venv tools by full path (`/content/venv/bin/python`, `/content/venv/bin/hf`).
Keep VS Code open and the machine awake during long runs. A runtime reset wipes `/content`.

Never paste the HF token into a file or a saved cell: use a Colab Secret named `HF_TOKEN`, or the `getpass` prompt.

def frame(t, shape=(4, 6, 3), person=0):
    return Frame(SimpleNamespace(shape=shape, person=person), t, None, 0)


class FakeInfer:
    """Counts persons from the fake image; remembers each batch it was given."""

    def __init__(self, fail_on_call=None, oom_above=None):
        self.batches = []
        self.ok_batches = []
        self.fail_on_call = fail_on_call
        self.oom_above = oom_above  # raise a fake CUDA OOM for batches larger than this

    def __call__(self, images):
        self.batches.append([im.shape for im in images])
        if self.fail_on_call == len(self.batches):
            raise RuntimeError("some other model failure")
        if self.oom_above is not None and len(images) > self.oom_above:
            raise RuntimeError("CUDA out of memory. Tried to allocate 2.00 GiB")
        self.ok_batches.append(list(images))
        return [([im.person, 0, 0, 0, 0], [0.9, 0, 0, 0, 0]) for im in images]


class ProcessFramesTest(unittest.TestCase):
    def test_batching(self):
        fi = FakeInfer()
        frames = [frame(F(i, 10)) for i in range(7)]
        records, error = run.process_frames(iter(frames), fi, 3, run.Timers())
        self.assertIsNone(error)
        self.assertEqual(len(records), 7)
        self.assertEqual([len(b) for b in fi.batches], [3, 3, 1])

    def test_flush_on_shape_change(self):
        fi = FakeInfer()
        shapes = [(4, 6, 3)] * 2 + [(6, 4, 3)] * 2 + [(4, 6, 3)]
        frames = [frame(F(i, 10), shape=s) for i, s in enumerate(shapes)]
        records, _ = run.process_frames(iter(frames), fi, 32, run.Timers(), keep_details=True)
        self.assertEqual([len(b) for b in fi.batches], [2, 2, 1])
        self.assertEqual([r.shape for r in records], shapes)
