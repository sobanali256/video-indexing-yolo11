    23	    def __init__(self):
    24	        self.files = {}
    25	        self.commits = 0

    27	    def list_repo_files(self, repo, repo_type=None):
    28	        return sorted(self.files)

    30	    def upload_folder(self, folder_path, path_in_repo, repo_id, repo_type, commit_message):
    31	        self.commits += 1
    32	        root = Path(folder_path)
    33	        for p in root.rglob("*"):
    34	            if p.is_file():
    35	                self.files[f"{path_in_repo}/{p.relative_to(root).as_posix()}"] = p.read_bytes()

    37	    def delete_folder(self, path_in_repo, repo_id, repo_type):
    38	        self.commits += 1
    39	        self.files = {k: v for k, v in self.files.items() if not k.startswith(path_in_repo + "/")}

    44	        target.write_bytes(self.files[filename])

class MainAbortTest(unittest.TestCase):
    def test_three_inference_failures_in_a_row_abort_without_writing(self):
        import tempfile
        from pathlib import Path

        import vidx.decode
        import vidx.model
        import vidx.source

        saved = (vidx.decode.decode_video, vidx.model.load_model, vidx.model.infer, vidx.source.iter_rows)

        def bad_infer(model, images):
            raise RuntimeError("driver broke")

        vidx.decode.decode_video = fake_decode()
        vidx.model.load_model = lambda: SimpleNamespace(device="cpu")
        vidx.model.infer = bad_infer
        vidx.source.iter_rows = lambda path: ((f"p:{i}", b"", None) for i in range(10))
        try:
            with tempfile.TemporaryDirectory() as tmp:
                (Path(tmp) / "a.parquet").touch()
                out = Path(tmp) / "r.jsonl"
                rc = run.main(["--input", tmp, "--output", str(out)])
                self.assertEqual(rc, 3)
                self.assertEqual(out.read_text(), "")
                self.assertEqual(Path(str(out) + ".done").read_text(), "")
        finally:
            (vidx.decode.decode_video, vidx.model.load_model, vidx.model.infer, vidx.source.iter_rows) = saved


def fake_decode_by_data(data):
    """Deterministic fake video: data is the video id as bytes; person count depends on it."""
    n = 2 + len(data) % 3
    for i in range(n):
        yield frame(F(i, 2), person=(len(data) + i) % 4)


class FakeRun:
    """Patches the heavy modules so run.main() works on fake videos; restores them afterwards."""

    def __init__(self, n_videos=8, crash_after=None, bad_infer=False):
        import tempfile

        self.n, self.crash_after, self.bad_infer = n_videos, crash_after, bad_infer
        self.tmp = tempfile.TemporaryDirectory()
        self.root = Path(self.tmp.name)
        (self.root / "in").mkdir()
        (self.root / "in" / "a.parquet").write_bytes(b"x" * 10)

    def rows(self, path):
        for i in range(self.n):
            if self.crash_after is not None and i >= self.crash_after:
                raise KeyboardInterrupt  # stands in for a kill; `finally` still runs
            yield f"a.parquet:{i}", f"video{i * 7}".encode(), None

    def __enter__(self):
        import vidx.decode
        import vidx.model
        import vidx.source

        self.mods = (vidx.decode, vidx.model, vidx.source)
        self.saved = (vidx.decode.decode_video, vidx.model.load_model, vidx.model.infer, vidx.source.iter_rows)
        vidx.decode.decode_video = fake_decode_by_data
        vidx.model.load_model = lambda: SimpleNamespace(device="cpu")
        vidx.model.infer = lambda model, images: FakeInfer()(images) if not self.bad_infer else 1 / 0
        vidx.source.iter_rows = self.rows
        return self

    def __exit__(self, *exc):
        d, m, s = self.mods
        d.decode_video, m.load_model, m.infer, s.iter_rows = self.saved
        self.tmp.cleanup()

    def main(self, out, *extra):
        return run.main(["--input", str(self.root / "in"), "--output", str(self.root / out), *extra])

    def text(self, out):
        return (self.root / out).read_text(encoding="utf-8")


class CheckpointMainTest(unittest.TestCase):
    def test_plain_run_does_not_import_checkpoint_module_or_write_extra_files(self):
        sys.modules.pop("vidx.checkpoint", None)
        with FakeRun() as fr:
            self.assertEqual(fr.main("r.jsonl"), 0)
            self.assertNotIn("vidx.checkpoint", sys.modules)
            self.assertEqual(sorted(p.name for p in fr.root.iterdir()), ["in", "r.jsonl", "r.jsonl.done"])

    def test_fresh_run_against_existing_checkpoint_exits_2(self):
        with FakeRun() as fr:
            ck = str(fr.root / "drive")
            self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck), 0)
            before = fr.text("r.jsonl")
            self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck), 2)
            self.assertEqual(fr.text("r.jsonl"), before)  # not overwritten
            self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck, "--run-name", "other"), 0)

    def test_resume_from_checkpoint_after_losing_local_files_matches_uninterrupted_run(self):
        # Parquet fixture: .venv has no pyarrow, so iter_rows and decode are faked (see FakeRun).
        with FakeRun(n_videos=8) as ref:
            self.assertEqual(ref.main("ref.jsonl"), 0)
            expected = sorted(ref.text("ref.jsonl").splitlines())
        with FakeRun(n_videos=8, crash_after=5) as fr:
            ck = str(fr.root / "drive")
            with self.assertRaises(KeyboardInterrupt):
                fr.main("r.jsonl", "--checkpoint-dir", ck, "--checkpoint-minutes", "0")
            (fr.root / "r.jsonl").unlink()  # runtime reset: local disk is gone
            (fr.root / "r.jsonl.done").unlink()
            fr.crash_after = None
            self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck, "--resume"), 0)
            self.assertEqual(sorted(fr.text("r.jsonl").splitlines()), expected)
            done = (fr.root / "r.jsonl.done").read_text().splitlines()
            self.assertEqual(len(done), 8)

    def test_snapshot_oserror_keeps_run_and_exit_code(self):
        import vidx.checkpoint as cp

        orig = cp.DirStore.publish

        def boom(self, staging, n):
            raise OSError("Drive disconnected")

        cp.DirStore.publish = boom
        try:
            with FakeRun() as fr:
                ck = str(fr.root / "drive")
                self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck, "--checkpoint-minutes", "0"), 0)
                self.assertEqual(len((fr.root / "r.jsonl.done").read_text().splitlines()), 8)
            with FakeRun(bad_infer=True) as fr:
                ck = str(fr.root / "drive")
                self.assertEqual(fr.main("r.jsonl", "--checkpoint-dir", ck), 3)  # abort code preserved
        finally:
            cp.DirStore.publish = orig

    def test_npz_temp_name_ends_with_npz_and_differs_from_final(self):

def lines_for(video, n):
    return [{"video": video, "second": s, "objects": [], "people": 0} for s in range(n)]


class OutputTest(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.out = Path(self.tmp.name) / "results.jsonl"

    def tearDown(self):
        self.tmp.cleanup()

    def read(self):
        return [json.loads(x) for x in self.out.read_text(encoding="utf-8").splitlines()]

    def test_write_and_sidecar(self):
        w = OutputWriter(self.out)
        w.write_video("a:0", lines_for("a:0", 2))
        w.write_video("a:1", [{"video": "a:1", "error": "bad"}])
        w.close()
        self.assertEqual(len(self.read()), 3)
        self.assertEqual(sidecar_path(self.out).read_text().split("\n"), ["a:0\t2", "a:1\t1", ""])

    def test_fresh_run_resets_old_files(self):
        w = OutputWriter(self.out)
        w.write_video("a:0", lines_for("a:0", 2))
        w.close()
        w = OutputWriter(self.out)  # no resume: start over
        w.close()
        self.assertEqual(self.out.read_text(), "")
        self.assertEqual(sidecar_path(self.out).read_text(), "")

    def test_resume_drops_half_written_video(self):
        w = OutputWriter(self.out)
        w.write_video("a:0", lines_for("a:0", 2))
        w.close()
        # crash: lines of a:1 reached the output (one truncated), sidecar never got a:1
        with open(self.out, "a", encoding="utf-8") as f:
            f.write(json.dumps(lines_for("a:1", 1)[0]) + "\n")
            f.write('{"video": "a:1", "sec')
        w = OutputWriter(self.out, resume=True)
        self.assertTrue(w.is_done("a:0"))
        self.assertFalse(w.is_done("a:1"))
        w.write_video("a:1", lines_for("a:1", 3))
        w.close()
        rows = self.read()
        self.assertEqual([r["video"] for r in rows], ["a:0", "a:0", "a:1", "a:1", "a:1"])

    def test_resume_redoes_video_whose_last_line_was_cut_off(self):
        w = OutputWriter(self.out)
        w.write_video("a:0", lines_for("a:0", 2))
        w.write_video("a:1", lines_for("a:1", 3))
        w.close()
        text = self.out.read_text(encoding="utf-8")
        self.out.write_text(text[:-10], encoding="utf-8")  # sidecar says a:1 is done, its last line is cut
        w = OutputWriter(self.out, resume=True)
        self.assertEqual(w.done, {"a:0"})
        w.close()
        self.assertEqual([r["video"] for r in self.read()], ["a:0", "a:0"])
        self.assertEqual(sidecar_path(self.out).read_text(), "a:0\t2\n")

    def test_resume_accepts_plain_sidecar_lines(self):
        self.out.write_text("".join(json.dumps(x) + "\n" for x in lines_for("a:0", 2)), encoding="utf-8")
        sidecar_path(self.out).write_text("a:0\n", encoding="utf-8")  # format of earlier runs
        w = OutputWriter(self.out, resume=True)
        self.assertEqual(w.done, {"a:0"})
        w.close()
        self.assertEqual(len(self.read()), 2)
        self.assertEqual(sidecar_path(self.out).read_text(), "a:0\t2\n")  # upgraded with the count

    def test_resume_ignores_truncated_sidecar_entry(self):
        w = OutputWriter(self.out)
        w.write_video("a:1", lines_for("a:1", 1))
        w.close()
        with open(sidecar_path(self.out), "a", encoding="utf-8") as f:
            f.write("a:1")  # "a:1x" truncated mid-write, no newline
        with open(self.out, "a", encoding="utf-8") as f:
            f.write(json.dumps(lines_for("a:12", 1)[0]) + "\n")
        w = OutputWriter(self.out, resume=True)
        self.assertEqual(w.done, {"a:1"})
        w.close()
        self.assertEqual([r["video"] for r in self.read()], ["a:1"])

    def test_resume_with_output_but_no_sidecar_refuses(self):
        w = OutputWriter(self.out)
        w.write_video("a:0", lines_for("a:0", 2))
        w.close()
        sidecar_path(self.out).unlink()
        before = self.out.read_text()
        with self.assertRaises(ResumeError) as cm:
            OutputWriter(self.out, resume=True)
        self.assertIn("sidecar", str(cm.exception))
        self.assertEqual(self.out.read_text(), before)  # not wiped
        OutputWriter(self.out).close()  # without --resume it starts fresh

    def test_resume_with_empty_output_and_no_sidecar_is_fine(self):
        self.out.write_text("")
        w = OutputWriter(self.out, resume=True)
        self.assertEqual(w.done, set())
        w.close()

    def test_resume_without_existing_files(self):
        w = OutputWriter(self.out, resume=True)
        self.assertEqual(w.done, set())
        w.close()
